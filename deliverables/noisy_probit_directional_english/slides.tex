\begin{frame}{Question and primary result}
Does separating the error correlations of true states 01 and 10 improve the five-parameter model and retain Full-channel performance with fewer audits?
\begin{itemize}
\item Six measurement parameters, compared with five for the pooled model and twelve for Full.
\item Equal-budget RMSE ratio: \textbf{\SameRatio}, 95\% paired interval \textbf{\SameCI}. Retention rule: \textbf{\SameResult}.
\item Half-budget RMSE ratio: \textbf{\HalfRatio}, interval \textbf{\HalfCI}. Retention rule: \textbf{\HalfResult}.
\end{itemize}
\source{Primary estimand: structural latent correlation $\rho$. The frozen rule also requires a coverage lower bound of at least 90\% and empirical numerical failures at most 1\%. A passed tolerance rule does not establish exact equivalence or universal audit savings.}
\end{frame}

\begin{frame}{The six-parameter conditional error channel}
For true state $t\in\{00,01,10,11\}$, let $r_0=\alpha$, $r_1=\gamma$ and
\[
E_k=\mathbf1\{Z_k\leq\Phi^{-1}(r_{t_k})\},\qquad
Z\mid Y=t\sim N_2\!\left(0,\begin{bmatrix}1&\kappa_t\\\kappa_t&1\end{bmatrix}\right).
\]
With $a_k=2(t_k\oplus o_k)-1$, the channel is
\[
C_{t,o}=\Phi_2\!\left(a_1\Phi^{-1}(r_{t_1}),a_2\Phi^{-1}(r_{t_2});a_1a_2\kappa_t\right).
\]
\begin{itemize}
\item Six parameters: $\alpha,\gamma,\kappa_{00},\kappa_{01},\kappa_{10},\kappa_{11}$.
\item $\kappa_{01}=\kappa_{10}$ recovers Five exactly.
\item Shared FP/FN margins remain a restriction. Four free row couplings span their interior binary distributions.
\end{itemize}
\source{Total fitted dimensions: Five 9, Six 10, Full 16. FP/FN rates lie in $(0,.5)$. Latent Gaussian $\kappa_t$ is not the binary error correlation.}
\end{frame}

\begin{frame}{Interpretation and limits}
\input{data/interpretation_slide.tex}
\begin{itemize}
\item Shared marginal error rates, homogeneous channels, and a meaningful fixed direction convention remain assumptions.
\item Unordered dyads are independent, and audits are random and error-free.
\item Fresh seeds test replication of these mechanisms, not external validity on real networks.
\end{itemize}
\source{Collapsed reciprocal-probit likelihood, mapped transformed-coordinate modes, and joint-Hessian intervals. No validation-driven estimator selection. All sensitivities and unsuccessful diagnostic attempts remain available.}
\end{frame}
